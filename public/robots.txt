# https://www.screna.ai/robots.txt
#
# Whitelist model: crawling is disallowed by default (Disallow: /), and only
# the marketing/content pages below are opened back up. Everything behind the
# login wall — /dashboard, /profile, /settings, /billing, /admin, /auth, the
# interview and onboarding flows, /coaching — is therefore blocked without
# needing its own rule.
#
# /interview-insights is deliberately NOT opened up: it is the personal-centre
# view of the interview-notes library. The same notes are served publicly at
# /interview-questions, which is the path allowed below. Opening both would
# have the two competing for the same queries.
#
# Per RFC 9309 the longest matching rule wins and Allow breaks ties, so each
# Allow below overrides the catch-all Disallow.
#
# The Allow list mirrors PRERENDER_STATIC in scripts/routes.mjs, which
# scripts/sitemap.mjs also uses for its static entries; src/test/vercel-routes.test.ts
# asserts they stay in sync.
#
# ── One group, on purpose ──
#
# There used to be a second group for the social-preview crawlers (LinkedIn,
# facebookexternalhit, Slackbot…), because middleware.ts answered exactly those
# user agents with a hand-built Open Graph document and the catch-all Disallow
# would have stopped them fetching at all. Both are gone: /experience/:id is
# server-rendered by api/experience/[id].ts and carries the og:/twitter: tags
# itself, so every agent gets the same document and no agent needs rules of its
# own. Per RFC 9309 §2.2.1 a crawler obeys only the most specific group matching
# its token, so a second group was also a second copy of every rule below —
# divergence waiting to happen, with dead link previews as the symptom.

User-agent: *

# Landing page. The $ anchors the match, so this is the home page only —
# without it, "/" would prefix-match every URL on the site.
Allow: /$

# Blog index and every post (/blog/:slug).
Allow: /blog

# Public "Interview Questions" surface: the company directory, every company
# page under it (/interview-questions/:companyId) and the crawlable pagination
# under those (/interview-questions/:companyId/page/:n), all fully readable
# logged out. Prefix match, so one rule covers all three.
#
# No $ anchor: `Allow: /interview-questions$` would open the directory only and
# leave the company pages to the catch-all Disallow — and the company pages are
# the ones carrying the question text worth indexing.
#
Allow: /interview-questions

# Single notes. /experience/:id is the only page carrying a whole write-up —
# the company pages show one trimmed sentence per note — so it is the page
# worth indexing, and this opens it.
#
# It does NOT mean every note gets indexed. Prefix rules cannot say "only the
# substantial ones", so the page itself sends noindex unless the note clears the
# content gate (isIndexablePost in scripts/routes.mjs) and its rendered body
# clears the word floor.
#
# /add-experience is unaffected — it does not start with "/experience".
Allow: /experience

# Marketing, support and legal pages.
Allow: /faq
Allow: /help
Allow: /contact
Allow: /privacy
Allow: /terms
Allow: /cookies
Allow: /data-protection

# Crawlers must be able to fetch the sitemap itself, otherwise Search Console
# reports it as unreadable. /sitemap.xml is a sitemap INDEX now, pointing at
# /sitemap-static.xml, /sitemap-companies.xml and one /sitemap-notes-YYYY-MM.xml
# per month — none of which the rule above covers, because it is anchored by the
# ".xml" it ends with rather than being a prefix. Without the wildcard line the
# index is readable and every file it names is blocked, which Search Console
# reports as "Couldn't fetch" against the children and costs every URL in them.
Allow: /sitemap.xml
Allow: /sitemap-*.xml
Allow: /llms.txt

# Rendering resources. Google renders pages before indexing them; blocking the
# CSS/JS bundles or the hero images would make the landing page look broken to
# the crawler. og-image.png is here too — an unfetchable image is an empty
# preview card on every shared link.
Allow: /*.css
Allow: /*.js
Allow: /*.png
Allow: /*.jpg
Allow: /*.jpeg
Allow: /*.webp
Allow: /*.svg
Allow: /*.ico
Allow: /*.woff2
Allow: /*.webmanifest

Disallow: /

Sitemap: https://www.screna.ai/sitemap.xml
